function validId(id) {
  return typeof id === 'string' && id.length > 0 && id.length <= 64 && !/[^A-Za-z0-9_-]/.test(id);
}

function validTitle(title) {
  return typeof title === 'string' && title.trim().length > 0;
}

function validTask(task) {
  return task !== null && typeof task === 'object' && !Array.isArray(task) &&
    validId(task.id) && validTitle(task.title) && typeof task.done === 'boolean';
}

export function createTask(id, title) {
  if (!validId(id) || !validTitle(title)) throw new TypeError('Invalid task ID or title');
  return {id, title, done: false};
}

export function completeTask(task) {
  if (!validTask(task)) throw new TypeError('Invalid task object');
  return {id: task.id, title: task.title, done: true};
}

export function filterOpen(tasks) {
  if (!Array.isArray(tasks) || !tasks.every(validTask)) throw new TypeError('Invalid task list');
  return tasks.filter(task => !task.done);
}
